This app exists to make itself unnecessary.

Every feature, every interaction, every line of code serves one purpose: **to help humans wake up from the dream of separation and live as conscious beings.**


### 13.2 Detection Algorithm

```typescript
interface DetectionResult {
  detected_markers: MarkerEvidence[];
  current_level_confidence: number;
  next_level_readiness: number;
  progression_recommendation: 'stay' | 'advance' | 'review';
}

async function detectVLevelMarkers(
  userId: string,
  currentVLevel: number,
  recentTexts: Array<{ text: string; source: string; timestamp: string }>
): Promise<DetectionResult> {
  
  const currentStage = Math.floor(currentVLevel);
  const nextStage = currentStage + 1;
  const transitionKey = `${currentStage}→${nextStage}`;
  
  const relevantMarkers = V_LEVEL_MARKERS[transitionKey] || [];
  const detectedMarkers: MarkerEvidence[] = [];
  
  // Scan all recent texts
  for (const { text, source, timestamp } of recentTexts) {
    for (const marker of relevantMarkers) {
      // Check regex patterns
      for (const pattern of marker.detection_patterns) {
        const match = text.match(pattern);
        if (match) {
          detectedMarkers.push({
            marker_id: marker.marker_id,
            text: match[0],
            source: source as any,
            detected_at: timestamp,
            confidence: marker.weight,
          });
        }
      }
    }
  }
  
  // Calculate readiness for next level
  const uniqueMarkers = [...new Set(detectedMarkers.map(m => m.marker_id))];
  const totalPossibleMarkers = relevantMarkers.length;
  const markerCoverage = uniqueMarkers.length / totalPossibleMarkers;
  
  const averageConfidence = detectedMarkers.length > 0
    ? detectedMarkers.reduce((sum, m) => sum + m.confidence, 0) / detectedMarkers.length
    : 0;
  
  const nextLevelReadiness = (markerCoverage * 0.6) + (averageConfidence * 0.4);
  
  // Determine recommendation
  let recommendation: 'stay' | 'advance' | 'review' = 'stay';
  if (nextLevelReadiness >= 0.8) {
    recommendation = 'advance';
  } else if (nextLevelReadiness < 0.3) {
    recommendation = 'review';
  }
  
  return {
    detected_markers: detectedMarkers,
    current_level_confidence: 1 - nextLevelReadiness,
    next_level_readiness: nextLevelReadiness,
    progression_recommendation: recommendation,
  };
}
```

```typescript
interface ProgressionCheck {
  can_progress: boolean;
  blocking_reasons: string[];
  progress_percentage: number;
  next_steps: string[];
}

async function checkVLevelProgression(
  userId: string,
  currentVLevel: number
): Promise<ProgressionCheck> {
  
  // Get user data
  const userData = await getUserProgressData(userId);
  const {
    days_at_level,
    tasks_completed,
    practice_minutes,
    streak,
    capacities,
    detected_markers
  } = userData;
  
  // Get criteria for current transition
  const transitionKey = getTransitionKey(currentVLevel);
  const criteria = PROGRESSION_CRITERIA[transitionKey];
  
  const blockingReasons: string[] = [];
  let metricsSum = 0;
  let metricsCount = 0;
  
  // Check days
  if (days_at_level < criteria.required_days) {
    blockingReasons.push(
      `Need ${criteria.required_days - days_at_level} more days at this level`
    );
  }
  metricsSum += Math.min(days_at_level / criteria.required_days, 1);
  metricsCount++;
  
  // Check tasks
  if (tasks_completed < criteria.required_tasks) {
    blockingReasons.push(
      `Complete ${criteria.required_tasks - tasks_completed} more tasks`
    );
  }
  metricsSum += Math.min(tasks_completed / criteria.required_tasks, 1);
  metricsCount++;
  
  // Check practice
  if (practice_minutes < criteria.required_practice_minutes) {
    blockingReasons.push(
      `Need ${criteria.required_practice_minutes - practice_minutes} more practice minutes`
    );
  }
  metricsSum += Math.min(practice_minutes / criteria.required_practice_minutes, 1);
  metricsCount++;
  
  // Check markers
  for (const requiredMarker of criteria.required_markers) {
    const markerCount = detected_markers.filter(
      m => m.marker_id === requiredMarker.id
    ).length;
    if (markerCount < requiredMarker.min_count) {
      blockingReasons.push(
        `Need more evidence of "${requiredMarker.id}"`
      );
    }
    metricsSum += Math.min(markerCount / requiredMarker.min_count, 1);
    metricsCount++;
  }
  
  // Check gate requirements if present
  if (criteria.gate_requirements) {
    for (const gate of criteria.gate_requirements) {
      if (!checkGate(gate, userData)) {
        blockingReasons.push(`Gate not met: ${gate}`);
      }
    }
  }
  
  const progressPercentage = (metricsSum / metricsCount) * 100;
  const canProgress = blockingReasons.length === 0;
  
  return {
    can_progress: canProgress,
    blocking_reasons: blockingReasons,
    progress_percentage: progressPercentage,
    next_steps: generateNextSteps(blockingReasons),
  };
}

async function advanceVLevel(userId: string): Promise<void> {
  const { v_level } = await getCurrentVLevel(userId);
  
  const currentSublevel = Math.round((v_level % 1) * 10);
  
  let newVLevel: number;
  if (currentSublevel === 3) {
    // Move to next main level
    newVLevel = Math.floor(v_level) + 1 + 0.1;
  } else {
    // Move to next sublevel
    newVLevel = Math.floor(v_level) + (currentSublevel + 1) / 10;
  }
  
  // Update database
  await supabase
    .from('adaptive_v_levels')
    .update({
      v_level: newVLevel,
      v_level_name: V_LEVEL_NAMES[newVLevel],
      previous_level: v_level,
      level_entered_at: new Date().toISOString(),
      days_at_level: 0,
      progression_markers: [],
    })
    .eq('user_id', userId);
  
  // Log progression
  await logVLevelProgression(userId, v_level, newVLevel);
}
```

```sql
-- User Capacities Table (already exists, showing complete structure)
CREATE TABLE user_capacities (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users NOT NULL UNIQUE,
  
  -- Awareness Cluster (1-10 scale)
  observation DECIMAL(3,1) DEFAULT 1,
  attention DECIMAL(3,1) DEFAULT 1,
  presence DECIMAL(3,1) DEFAULT 1,
  
  -- Inquiry Cluster
  self_inquiry DECIMAL(3,1) DEFAULT 1,
  disidentification DECIMAL(3,1) DEFAULT 1,
  pattern_recognition DECIMAL(3,1) DEFAULT 1,
  
  -- Embodiment Cluster
  body_awareness DECIMAL(3,1) DEFAULT 1,
  emotional_mastery DECIMAL(3,1) DEFAULT 1,
  equanimity DECIMAL(3,1) DEFAULT 1,
  
  -- Integration Cluster
  shadow_integration DECIMAL(3,1) DEFAULT 1,
  surrender DECIMAL(3,1) DEFAULT 1,
  unification DECIMAL(3,1) DEFAULT 1,
  
  -- Metadata
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- Capacity Clusters View
CREATE OR REPLACE VIEW capacity_clusters AS
SELECT 
  user_id,
  
  -- Cluster Averages
  ROUND((observation + attention + presence) / 3.0, 1) AS awareness_cluster,
  ROUND((self_inquiry + disidentification + pattern_recognition) / 3.0, 1) AS inquiry_cluster,
  ROUND((body_awareness + emotional_mastery + equanimity) / 3.0, 1) AS embodiment_cluster,
  ROUND((shadow_integration + surrender + unification) / 3.0, 1) AS integration_cluster,
  
  -- Overall H-Level (average of all 12)
  ROUND(
    (observation + attention + presence + 
     self_inquiry + disidentification + pattern_recognition +
     body_awareness + emotional_mastery + equanimity +
     shadow_integration + surrender + unification) / 12.0, 
    1
  ) AS h_level,
  
  -- Individual capacities for display
  observation, attention, presence,
  self_inquiry, disidentification, pattern_recognition,
  body_awareness, emotional_mastery, equanimity,
  shadow_integration, surrender, unification
  
FROM user_capacities;

-- Capacity History for tracking changes over time
CREATE TABLE capacity_history (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users NOT NULL,
  capacity_name TEXT NOT NULL,
  old_value DECIMAL(3,1),
  new_value DECIMAL(3,1),
  change_reason TEXT, -- 'task_completion', 'ai_detection', 'manual'
  evidence JSONB, -- supporting evidence for the change
  changed_at TIMESTAMPTZ DEFAULT now()
);

-- Enable RLS
ALTER TABLE user_capacities ENABLE ROW LEVEL SECURITY;
ALTER TABLE capacity_history ENABLE ROW LEVEL SECURITY;

-- RLS Policies
CREATE POLICY "Users can view own capacities"
  ON user_capacities FOR SELECT
  USING (auth.uid() = user_id);

CREATE POLICY "Users can update own capacities"
  ON user_capacities FOR UPDATE
  USING (auth.uid() = user_id);

CREATE POLICY "Users can view own capacity history"
  ON capacity_history FOR SELECT
  USING (auth.uid() = user_id);
```

### 21.2 Capacity Update Algorithm

```typescript
interface CapacityUpdate {
  capacity_name: string;
  delta: number;
  reason: string;
  evidence?: string;
}

async function updateCapacity(
  userId: string,
  updates: CapacityUpdate[]
): Promise<void> {
  
  // Get current values
  const { data: current } = await supabase
    .from('user_capacities')
    .select('*')
    .eq('user_id', userId)
    .single();
  
  // Apply updates with modifiers
  const newValues: Record<string, number> = {};
  
  for (const update of updates) {
    const currentValue = current[update.capacity_name.toLowerCase()];
    
    // Apply growth rate modifier
    let modifier = 1.0;
    if (currentValue < 3) modifier = 1.5;
    else if (currentValue >= 7) modifier = 0.5;
    else if (currentValue >= 9) modifier = 0.25;
    
    const adjustedDelta = update.delta * modifier;
    const newValue = Math.min(10, currentValue + adjustedDelta);
    
    newValues[update.capacity_name.toLowerCase()] = newValue;
    
    // Log to history
    await supabase.from('capacity_history').insert({
      user_id: userId,
      capacity_name: update.capacity_name,
      old_value: currentValue,
      new_value: newValue,
      change_reason: update.reason,
      evidence: update.evidence ? { text: update.evidence } : null,
    });
  }
  
  // Update capacities
  await supabase
    .from('user_capacities')
    .update({ ...newValues, updated_at: new Date().toISOString() })
    .eq('user_id', userId);
}
```
